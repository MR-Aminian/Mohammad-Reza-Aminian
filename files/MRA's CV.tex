\section{\textbf{Research}}

\cvline{}{\textbf{\large Fundraising for Education: Optimizing Crowdfunding Platforms} \textnormal{\normalsize (Job Market Paper)} \newline Design dynamic assortment policies for crowdfunding platforms to prevent ``project starvation'' and maximize the value of fully funded campaigns. \newline {with} Rad Niazadeh (Chicago Booth)}
\vspace{0.1 cm}

\cvline{}{\textbf{\large Optimal Bayesian Online Allocation of Reusable Resources}
\newline Generalize classical online resource allocation theory to the reusable setting by developing a time-invariant admission policy that guarantees ex-post feasibility and matches the information-theoretic tight bounds of non-reusable models.  \href{https://papers.ssrn.com/sol3/papers.cfm?abstract_id=6384158}{SSRN: 6384158}
\newline {with} Rad Niazadeh (Chicago Booth), Pranav Nuti (Chicago Booth) }
\vspace{0.1 cm}

\cvline{}{\textbf{\large Stationary Online Contention Resolution Schemes}
\newline Create the S-OCRS framework to unify the design of online selection algorithms, achieving optimal selectability and resolving open problems across various feasibility environments, like matchings and matroids. \href{https://papers.ssrn.com/sol3/papers.cfm?abstract_id=6384199}{SSRN: 6384199}
\newline {with} Rad Niazadeh (Chicago Booth), Pranav Nuti (Chicago Booth)
\newline {Major Revision at Operations Research, Accepted at EC 2026}}
\vspace{-0.15 cm}

\cvline{}{$\bullet$ Finalist, Michael H. Rothkopf Junior Researcher Paper Prize, \textit{INFORMS Auctions and Market Design (AMD) Section (Entrant: Pranav Nuti)}}
\vspace{0.25 cm}

\cvline{}{\textbf{\large Online Job Selection: Reward Rate vs. Remaining Value} (formerly Real-Time Personalized Order Holding) \newline  Introduce an online resource allocation framework and index-based policies with theoretical guarantees to optimize the trade-off between keeping an ongoing job to harvest its remaining value vs. preempting it to serve a new request in limited capacity systems. \href{https://ssrn.com/abstract=4644495}{SSRN: 4644495}
\newline {with}  Will Ma (Columbia GSB), Linwei Xin (Cornell)
\newline {Accepted at Management Science}}
